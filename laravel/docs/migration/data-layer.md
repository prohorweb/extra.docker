App\\Data\\Contracts;\n\nuse Illuminate\\Database\\Eloquent\\Model;\n\ninterface DataContract\n{\n    /**\n     * Создаёт DTO из Eloquent модели\n     */\n    public static function fromModel(Model $model): static;\n\n    /**\n     * Создаёт коллекцию DTO из коллекции моделей\n     */\n    public static function collection(Iterable $models): array;\n\n    /**\n     * Преобразует в массив для JSON/API\n     */\n    public function toArray(): array;\n}\n```\n\n---\n\n## Базовый абстрактный DTO\n\n**Файл**: `app/Data/BaseData.php`\n\n```php\n<?php\n\nnamespace App\\Data;\n\nuse App\\Data\\Contracts\\DataContract;\nuse Illuminate\\Database\\Eloquent\\Model;\nuse Illuminate\\Support\\Collection;\n\nabstract readonly class BaseData implements DataContract\n{\n    public static function collection(Iterable $models): array\n    {\n        return collect($models)->map(fn($m) => static::fromModel($m))->toArray();\n    }\n\n    public function toArray(): array\n    {\n        return get_object_vars($this);\n    }\n}\n```\n\n---\n\n## Пример: ServiceCardData\n\n**Файл**: `app/Data/Services/ServiceCardData.php`\n\n```php\n<?php\n\nnamespace App\\Data\\Services;\n\nuse App\\Data\\BaseData;\nuse App\\Models\\Service;\nuse Illuminate\\Support\\Str;\n\nreadonly class ServiceCardData extends BaseData\n{\n    public function __construct(\n        public int $id,\n        public string $title,\n        public string $slug,\n        public string $shortDescription,\n        public string $imageUrl,\n        public string $priceFrom,\n        public array $tags = [],\n        public ?string $categoryName = null,\n    ) {}\n\n    public static function fromModel(Service $service): self\n    {\n        return new self(\n            id: $service->id,\n            title: $service->title,\n            slug: $service->slug,\n            shortDescription: Str::limit(strip_tags($service->description), 120),\n            imageUrl: $service->getFirstMediaUrl('card') ?? asset('images/placeholder-card.jpg'),\n            priceFrom: $service->price_from\n                ? number_format($service->price_from, 0, ',', ' ') . ' ₽'\n                : 'По запросу',\n            tags: $service->tags->pluck('name')->toArray(),\n            categoryName: $service->category?->title,\n        );\n    }\n}\n```\n\n---\n\n## Пример: ServiceDetailData\n\n**Файл**: `app/Data/Services/ServiceDetailData.php`\n\n```php\n<?php\n\nnamespace App\\Data\\Services;\n\nuse App\\Data\\BaseData;\nuse App\\Models\\Service;\n\nreadonly class ServiceDetailData extends BaseData\n{\n    public function __construct(\n        public int $id,\n        public string $title,\n        public string $slug,\n        public string $description,\n        public array $images = [],\n        public array $features = [],\n        public array $benefits = [],\n        public ?string $priceFrom = null,\n        public ?string $priceTo = null,\n        public array $tags = [],\n        public ?string $categoryName = null,\n        public array $relatedServices = [],\n        public SEOData $seo,\n    ) {}\n\n    public static function fromModel(Service $service): self\n    {\n        $service->load(['media', 'tags', 'category', 'features', 'benefits', 'relatedServices']);\n\n        return new self(\n            id: $service->id,\n            title: $service->title,\n            slug: $service->slug,\n            description: $service->description,\n            images: $service->getMedia('images')->map(fn($m) => [\n                'url' => $m->getUrl(),\n                'thumb' => $m->getUrl('thumb'),\n                'alt' => $m->custom_properties['alt'] ?? $service->title,\n            ])->toArray(),\n            features: $service->features->pluck('title')->toArray(),\n            benefits: $service->benefits->pluck('title')->toArray(),\n            priceFrom: $service->price_from ? number_format($service->price_from, 0, ',', ' ') . ' ₽' : null,\n            priceTo: $service->price_to ? number_format($service->price_to, 0, ',', ' ') . ' ₽' : null,\n            tags: $service->tags->pluck('name')->toArray(),\n            categoryName: $service->category?->title,\n            relatedServices: ServiceCardData::collection($service->relatedServices),\n            seo: SEOData::fromModel($service),\n        );\n    }\n}\n```\n\n---\n\n## Service Layer Pattern\n\n**Файл**: `app/Services/ServiceService.php`\n\n```php\n<?php\n\nnamespace App\\Services;\n\nuse App\\Data\\Services\\ServiceCardData;\nuse App\\Data\\Services\\ServiceDetailData;\nuse App\\Data\\Services\\ServiceFilterData;\nuse App\\Models\\Service;\nuse Illuminate\\Contracts\\Pagination\\LengthAwarePaginator;\nuse Illuminate\\Support\\Collection;\n\nclass ServiceService\n{\n    public function getFeatured(int $limit = 6): Collection\n    {\n        return Service::query()\n            ->active()\n            ->featured()\n            ->with(['media', 'category'])\n            ->orderBy('position')\n            ->limit($limit)\n            ->get()\n            ->map(fn($s) => ServiceCardData::fromModel($s));\n    }\n\n    public function getPaginated(ServiceFilterData $filter): LengthAwarePaginator\n    {\n        return Service::query()\n            ->active()\n            ->with(['media', 'category'])\n            ->when($filter->categoryId, fn($q) => $q->where('category_id', $filter->categoryId))\n            ->when($filter->search, fn($q) => $q->where('title', 'like', \"%{$filter->search}%\"))\n            ->when($filter->tag, fn($q) => $q->whereHas('tags', fn($t) => $t->where('slug', $filter->tag)))\n            ->orderBy('position')\n            ->paginate($filter->perPage ?? 12)\n            ->through(fn($s) => ServiceCardData::fromModel($s));\n    }\n\n    public function getBySlug(string $slug): ?ServiceDetailData\n    {\n        $service = Service::query()\n            ->active()\n            ->with(['media', 'tags', 'category', 'features', 'benefits', 'relatedServices'])\n            ->where('slug', $slug)\n            ->first();\n\n        return $service ? ServiceDetailData::fromModel($service) : null;\n    }\n\n    public function getRelated(Service $service, int $limit = 3): Collection\n    {\n        return $service->relatedServices()\n            ->active()\n            ->with('media')\n            ->limit($limit)\n            ->get()\n            ->map(fn($s) => ServiceCardData::fromModel($s));\n    }\n}\n```\n\n---\n\n## Filter Data Illuminate\\View\\View;\n\nclass ServiceController extends Controller\n{\n    public function __construct(\n        private readonly ServiceService $serviceService\n    ) {}\n\n    public function index(ServiceFilterRequest $request): View\n    {\n        $filter = ServiceFilterData::fromRequest($request->validated());\n        $services = $this->serviceService->getPaginated($filter);\n        $categories = Category::active()->get();\n\n        return view('pages.services.index', [\n            'services' => $services,\n            'categories' => $categories,\n            'filter' => $filter,\n            'seo' => SEOData::forServicesIndex(),\n        ]);\n    }\n\n    public function show(string $slug): View\n    {\n        $service = $this->serviceService->getBySlug($slug);\n\n Правила Data Layer\n\n| Правило | Описание |\n|---------|----------|\n| **Immutable** | Все DTO — `readonly class` |\n| **Type-safe** | Строгая типизация всех полей |\n| **Single Responsibility** | CardData для списков, DetailData для деталей, FilterData для фильтров |\n| **No Logic in DTO** | Только данные, никакой бизнес-логики |\n| **Service = Aggregation** | Сервисы агрегируют данные из моделей в DTO |\n| **Blade gets DTO Only** | Шаблоны получают только Data-объекты |\n| **fromModel Static** | Фабричный метод для создания из Eloquent |\n| **Collection Helper** | `::collection()` для пакетного преобразования |\n\n---\n\n## Навигация\n\n- [← Roadmap](./roadmap.md)\n- [Foundation →](./foundation.md)\n- [Components →](./components.md)\n- [Patterns →](./patterns.md)"
